# 灵动岛小组件代码

双岛 Flex 联动布局：左侧番茄计时器主岛（可展开），右侧音乐副岛（点击控制网易云播放器）。

## 三、JavaScript

```js
// ==========================================
// 灵动岛小组件
// ==========================================
var widget = document.getElementById('floating-widget');
var diTimer = document.getElementById('di-timer');
var diPill = document.getElementById('di-pill');
var diPanel = document.getElementById('di-panel');
var diTime = document.getElementById('di-time');
var diBigTime = document.getElementById('di-big-time');
var diStatus = document.getElementById('di-status');
var diBarFill = document.getElementById('di-bar-fill');
var diPlayBtn = document.getElementById('di-play-btn');
var diResetBtn = document.getElementById('di-reset-btn');
var diWorkBtn = document.getElementById('di-work-btn');
var diBreakBtn = document.getElementById('di-break-btn');
var diMusicPill = document.getElementById('di-music-pill');

var diExpanded = false;
var diIsRunning = false;
var diMode = 'work'; // work | break
var diAnimating = false;
var diMusicPlaying = false;
var diNeteasePlaying = false;
var diNeteasePanel = document.getElementById('netease-panel');
var diNeteaseIframe = document.getElementById('netease-iframe-di');

// 显示/隐藏
function _showFloatingWidget() {
    widget.classList.add('visible');
    diExpanded = false;
    diTimer.classList.remove('expanded', 'expanding', 'collapsing');
    widget.classList.remove('has-expanded');
    diBarFill.style.width = '0%';
}
function _hideFloatingWidget() {
    widget.classList.remove('visible');
    diExpanded = false;
    diTimer.classList.remove('expanded', 'expanding', 'collapsing');
    widget.classList.remove('has-expanded');
}

// 展开/收起动画
function _diToggleExpand() {
    if (diAnimating) return;
    diAnimating = true;
    diExpanded = !diExpanded;
    if (diExpanded) {
        _closeNeteasePanel();
        diTimer.classList.remove('collapsing');
        diTimer.classList.add('expanding');
        widget.classList.add('has-expanded');
        // staggered row-in: last row delay 0.25s + duration 0.3s = 0.55s
        setTimeout(function() {
            diTimer.classList.remove('expanding');
            diTimer.classList.add('expanded');
            diAnimating = false;
        }, 580);
    } else {
        diTimer.classList.remove('expanding');
        diTimer.classList.add('collapsing');
        widget.classList.remove('has-expanded');
        // quick row-out: last row delay 0.06s + duration 0.1s = 0.16s
        // then container shrinks after rows are gone
        setTimeout(function() {
            diTimer.classList.remove('expanded', 'collapsing');
            diAnimating = false;
        }, 700);
    }
}

// 关闭网易云面板
function _closeNeteasePanel() {
    if (!diNeteasePlaying) return;
    diNeteaseIframe.src = '';
    diNeteasePanel.classList.remove('visible');
    diMusicPill.classList.remove('playing');
    diNeteasePlaying = false;
    try { pomodoroFrame.contentWindow.postMessage({type:'WIDGET_CMD',action:'music-netease-off'}, '*'); } catch(e) {}
}

// 接收计时器状态（由 pomodoro.html 通过 postMessage 发送）
function _onTimerState(data) {
    var timeLeft = data.timeLeft || 0;
    var min = Math.floor(timeLeft / 60);
    var sec = timeLeft % 60;
    var timeStr = String(min).padStart(2,'0') + ':' + String(sec).padStart(2,'0');
    diIsRunning = !!data.isRunning;
    diMode = data.mode || 'work';

    // 推算总时长
    var total = diMode === 'work' ? 25*60 : 5*60;
    try { var t = localStorage.getItem('pomodoro-settings'); if(t) { var s=JSON.parse(t); total = diMode==='work' ? (s.workTime||25)*60 : (s.breakTime||5)*60; } } catch(e){}

    // 更新时间
    diTime.textContent = timeStr;
    diBigTime.textContent = timeStr;
    // 状态文字
    diStatus.textContent = diIsRunning ? (diMode === 'work' ? '专注中' : '休息中') : '已暂停';
    // 播放按钮
    diPlayBtn.textContent = diIsRunning ? '⏸ 暂停' : '▶ 开始';
    // 进度条
    var progress = total > 0 ? (total - timeLeft) / total : 0;
    diBarFill.style.width = (progress * 100) + '%';
    // 模式按钮高亮
    diWorkBtn.classList.toggle('active', diMode === 'work');
    diBreakBtn.classList.toggle('active', diMode !== 'work');
    // 进度条颜色微调
    diBarFill.style.boxShadow = diMode === 'work' ? '0 0 6px rgba(255,120,80,0.25)' : '0 0 6px rgba(80,200,120,0.25)';
}

// 接收音乐状态（由 pomodoro.html 通过 postMessage 发送）
function _onMusicState(data) {
    diMusicPlaying = !!data.isPlaying;
    diMusicPill.classList.toggle('playing', diMusicPlaying);
}

// 按钮事件
diPill.addEventListener('click', function() { _diToggleExpand(); });
diPlayBtn.addEventListener('click', function(e) {
    e.stopPropagation();
    try { pomodoroFrame.contentWindow.postMessage({type:'WIDGET_CMD',action:'toggle'}, '*'); } catch(e) {}
});
diResetBtn.addEventListener('click', function(e) {
    e.stopPropagation();
    try { pomodoroFrame.contentWindow.postMessage({type:'WIDGET_CMD',action:'reset'}, '*'); } catch(e) {}
});
diWorkBtn.addEventListener('click', function(e) {
    e.stopPropagation();
    try { pomodoroFrame.contentWindow.postMessage({type:'WIDGET_CMD',action:'setTime',minutes:parseInt(document.getElementById('work-time')?.value||25)}, '*'); } catch(e) {}
    try { pomodoroFrame.contentWindow.postMessage({type:'WIDGET_CMD',action:'mode-work'}, '*'); } catch(e) {}
});
diBreakBtn.addEventListener('click', function(e) {
    e.stopPropagation();
    try { pomodoroFrame.contentWindow.postMessage({type:'WIDGET_CMD',action:'setTime',minutes:parseInt(document.getElementById('break-time')?.value||5)}, '*'); } catch(e) {}
    try { pomodoroFrame.contentWindow.postMessage({type:'WIDGET_CMD',action:'mode-break'}, '*'); } catch(e) {}
});

// 音乐岛点击 → 先收起计时器岛，再 toggle 网易云
diMusicPill.addEventListener('click', function() {
    if (diNeteasePlaying) {
        _closeNeteasePanel();
    } else {
        var wasExpanded = diExpanded;
        if (wasExpanded) _diToggleExpand();
        setTimeout(function() {
            var playlistId = localStorage.getItem('pomodoro-netease-playlist') || '9237474400';
            diNeteaseIframe.src = 'https://music.163.com/outchain/player?type=0&id=' + playlistId + '&auto=1&height=110';
            diNeteasePanel.classList.add('visible');
            diMusicPill.classList.add('playing');
            diNeteasePlaying = true;
            try { pomodoroFrame.contentWindow.postMessage({type:'WIDGET_CMD',action:'music-netease-on'}, '*'); } catch(e) {}
        }, wasExpanded ? 700 : 0);
    }
});

// 点击面板背景（非按钮）→ 收起
diPanel.addEventListener('click', function(e) {
    if (e.target.closest('.di-btn')) return;
    if (diExpanded) _diToggleExpand();
});

// postMessage 监听（灵动岛相关部分）
window.addEventListener('message', function(event) {
    if (event.data && event.data.type === 'TIMER_STATE') {
        _onTimerState(event.data);
    }
    else if (event.data && event.data.type === 'MUSIC_STATE') {
        _onMusicState(event.data);
    }
});
```

## 四、动画时序说明

### 展开（胶囊 → 方框）

| 时机 | 动作 |
|------|------|
| 0ms | 容器 `width` 开始从 ~90px → 316px（spring 曲线 `0.34,1.56,0.64,1`，0.5s），`border-radius` 从 999px → 22px |
| 0ms | 胶囊内容 `.di-pill` 淡出（opacity 0, scale 0.95, 0.15s） |
| 100ms | 面板第 1 行开始浮现（opacity 0→1, translateY 10px→0, 0.3s） |
| 150ms | 面板第 2 行开始浮现 |
| 200ms | 面板第 3 行开始浮现 |
| 250ms | 面板第 4 行开始浮现 |
| 550ms | 所有行动画完成 |
| 580ms | JS 切换 `expanding` → `expanded`，解锁 |

### 收起（方框 → 胶囊）

| 时机 | 动作 |
|------|------|
| 0ms | 面板所有行快速淡出（opacity 0, scale 0.95, 0.1s） |
| 180ms | 容器 `width` 清空，开始收缩回 ~90px（spring 曲线，0.5s） |
| 700ms | JS 清除所有状态类，解锁 |

### 音乐岛联动

- 主岛展开时，音乐岛 `scale(0.92)` + `opacity(0.7)`（spring 曲线，0.5s）
- 点击音乐岛时，先收起主岛（等待 700ms），再打开网易云浮窗
- 网易云浮窗固定在灵动岛下方 100px 处，独立于 iframe
