## 📈 学习进度概览

```dataviewjs
const pages = dv.pages('"B2口语素材"').where(p => p.ru && !p.file.name.includes('章节索引') && !p.file.name.includes('总索引'))
const bySection = {}
for (const p of pages) {
  const s = p.section || '未分类'
  bySection[s] = (bySection[s]||0) + 1
}

// Build bar chart as text
const total = pages.length
const order = ['场景对话', '核心句型', '词汇小灶', '表达技巧', '素材更新']
dv.list(order.map(s => {
  const count = bySection[s] || 0
  const pct = ((count/total)*100).toFixed(1)
  const bar = '█'.repeat(Math.round(count/total*50))
  return `${s}: ${count} 条 (${pct}%) ${bar}`
}))
```
